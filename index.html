Sync Script Setup (one-time)
1. Create a new Google Apps Script at script.google.com
2. Paste this code:
const SHEET_NAME="AKAI_CONFIG";
function doGet(e){
 var ss=SpreadsheetApp.getActiveSpreadsheet();
 var sh=ss.getSheetByName(SHEET_NAME)||ss.insertSheet(SHEET_NAME);
 var v=sh.getRange("A1").getValue();
 return ContentService.createTextOutput(v||"{}").setMimeType(ContentService.MimeType.JSON);
}
function doPost(e){
 var ss=SpreadsheetApp.getActiveSpreadsheet();
 var sh=ss.getSheetByName(SHEET_NAME)||ss.insertSheet(SHEET_NAME);
 var data=e.postData?e.postData.contents:"{}";
 sh.getRange("A1").setValue(data);
 return ContentService.createTextOutput("ok").setMimeType(ContentService.MimeType.TEXT);
}
3. Deploy as Web App → Execute as: Me → Who has access: Anyone
4. Copy the /exec URL and paste it in the Sync URL field above
5. Click “Save to Sheet” — now every phone that has this sync URL gets instant updates when they open the app
